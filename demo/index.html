# Eagle script with CLR object interop
proc greet {name {greeting "Hello"}} {
  puts "$greeting, $name!"
}

# Use CLR objects directly
set sb [object create -alias System.Text.StringBuilder]
$sb Append "Eagle "
$sb Append "on the CLR"
puts [$sb ToString]
object dispose $sb

# List operations
set items [list alpha bravo charlie delta]
foreach item [lsort -decreasing $items] {
  puts " $item"
}

# SQL database access
sql open -type SQLite "Data Source=:memory:"
sql execute "CREATE TABLE t1 (id INTEGER, name TEXT)"

local configs = require('lspconfig.configs')
configs.eagle = {
  default_config = {
    cmd = { 'node', '/path/to/eagle-lsp/server.js', '--stdio' },
    filetypes = { 'eagle', 'tcl' },
  }
}
require('lspconfig').eagle.setup({})
(lsp-register-client
 (make-lsp-client
  :new-connection (lsp-stdio-connection
   '("node" "/path/to/eagle-lsp/server.js" "--stdio"))
  :major-modes '(tcl-mode)
  :server-id 'eagle-ls))
{
  "clients": {
    "eagle": {
      "enabled": true,
      "command": ["node", "/path/to/server.js", "--stdio"],
      "selector": "source.tcl"
    }
  }
}
[language-server.eagle-ls]
command = "node"
args = ["/path/to/eagle-lsp/server.js", "--stdio"]

[[language]]
name = "eagle"

language-servers = ["eagle-ls"]
